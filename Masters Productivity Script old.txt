CREATE OR REPLACE TABLE SBX_PSAS_DB.SALES_OPS_GOV.T_MASTERS_ACCT_HIST_TASKS AS 
SELECT h.CUST_ACCT_ID AS accountnumber_HIST
    -- ,h.period AS HISTORY_PERIOD
	,h.ACCOUNT_OWNER AS History_Account_Owner
    ,H.USER_MANAGER
    ,a.owner_name__c AS Current_Account_Owner
	,a.name AS Account_Name
	,a.ID AS Account_ID
	,a.accountnumber AS accountnumber_CURR
    ,a.territory_id__c
	,t.whatid
	,t.id AS Task_ID
    ,t.dm_discussion__c
    ,t.ownerid AS Activity_Owner_ID
    ,t.subject
	,CASE WHEN t.dm_discussion__c = 'true'THEN 1 ELSE 0 END AS Discussion_with_decision_maker
	,t.activitydate
	,TO_VARCHAR(t.activitydate, 'YYYY-MM-01') AS ACTIVITY_PERIOD
    ,CONTACT.EMAIL AS CONTACT_EMAIL
    ,CONTACT.FULL_NAME__C AS CONTACT_NAME
    ,CONTACT.DECISION_MAKER__C AS CONTACT_DMC_CHECK_BOX
    ,T.CALLTYPE
    ,T.TASKSUBTYPE
    ,H.CAREER_LEVEL
    ,H.LEVEL_START_DATE
    ,H.LEVEL_END_DATE
    ,H.LEVEL_STATUS_DSCR
-- FROM SBX_PSAS_DB.SALES_OPS_GOV.V_MASTERS_SFDC_ACCT_ASSIGNMENTS_HIST h
FROM SBX_PSAS_DB.SALES_OPS_GOV.V_MASTERS_SFDC_ACCT_ASSIGNMENTS H
LEFT JOIN PRD_ENT_PL_DATALAKE_DB.PHARMA2_SFDC.Account a ON ltrim(h.CUST_ACCT_ID,'0') = ltrim(a.accountnumber,'0')
LEFT JOIN PRD_ENT_PL_DATALAKE_DB.PHARMA2_SFDC.V_TASK t ON ltrim(t.whatid,'0') = ltrim(a.ID,'0')
LEFT JOIN PRD_ENT_PL_DATALAKE_DB.PHARMA2_SFDC.V_CONTACT CONTACT ON LTRIM(CONTACT.ID,'0') = LTRIM(T.WHOID,'0')
    -- AND TO_VARCHAR(t.activitydate, 'YYYY-MM-01') = h.period
